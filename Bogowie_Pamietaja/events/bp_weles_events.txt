# A player explicitly picks which of THEIR living ADULT prisoners to offer.
# The target survives until the choice in this confirmation event.
bp_weles.2004 = {
    type = character_event
    theme = diplomacy
    override_background = {
        reference = bp_scene_weles_03
    }
    title = bp_weles.2004.t
    desc = bp_weles.2004.desc
    left_portrait = root
    right_portrait = scope:bp_offering_person
    trigger = {
        exists = var:bp_weles_proposed_prisoner
        var:bp_weles_proposed_prisoner = {
            is_alive = yes
            is_adult = yes
            is_imprisoned_by = root
        }
    }
    on_trigger_fail = {
        if = {
            limit = { has_variable = bp_weles_proposed_prisoner }
            remove_variable = bp_weles_proposed_prisoner
        }
    }
    immediate = {
        var:bp_weles_proposed_prisoner = { save_scope_as = bp_offering_person }
    }
    option = {
        name = bp_weles.2004.a
        if = {
            limit = { scope:bp_offering_person = { dynasty = root.dynasty } }
            add_character_flag = { flag = bp_weles_dynasty_blood years = 15 }
            dynasty = { set_variable = { name = bp_weles_dynastic_blood_v018 value = yes } }
            add_stress = 60
            add_tyranny = 20
        }
        # Survivors react to a real, named death. This is independent of dynasty status.
        scope:bp_offering_person = {
            every_close_family_member = {
                limit = {
                    is_alive = yes
                    NOT = { this = root }
                }
                add_opinion = { target = root modifier = bp_weles_relative_sacrifice_opinion }
            }
            death = { killer = root death_reason = death_execution }
        }
        remove_character_flag = bp_weles_waiting_for_prisoner
        remove_variable = bp_weles_proposed_prisoner
        add_character_flag = { flag = bp_weles_offering days = 90 }
        add_character_flag = { flag = bp_weles_blood days = 730 }
        dynasty = { set_variable = { name = bp_weles_prisoner_sacrificed_ever_v030 value = yes } }
        add_character_flag = { flag = bp_weles_blood_now_v020 days = 120 }
        bp_weles_plus_20_effect = yes
        add_dread = 15
        bp_040_actual_blood_effect = yes
        bp_040_schedule_revenge_effect = yes
        ai_chance = { base = 35 }
    }
    option = {
        name = bp_weles.2004.b
        remove_character_flag = bp_weles_waiting_for_prisoner
        remove_variable = bp_weles_proposed_prisoner
        add_prestige = -80
        add_character_flag = { flag = bp_weles_offering days = 90 }
        add_character_flag = { flag = bp_weles_merciful days = 730 }
        add_character_flag = { flag = bp_weles_mercy_now_v020 days = 120 }
        bp_weles_plus_10_effect = yes
        bp_040_mercy_effect = yes
        ai_chance = { base = 65 }
    }
}

    type = character_event
    theme = diplomacy

    override_background = { reference = bp_scene_deity_weles_2020 }

            triggered_desc = {
                trigger = { has_character_flag = bp_040_weles_bargain_v040 }

            triggered_desc = {
                trigger = {
                    dynasty = { has_variable = bp_weles_prisoner_sacrificed_ever_v030 }
                }
                desc = bp_weles.2020.desc_blood
            }
            triggered_desc = {
                trigger = { has_trait = brave }
                desc = bp_weles.2020.desc_brave

                desc = bp_weles.2020.desc_craven

            desc = bp_weles.2020.desc_default
        }
    }
    left_portrait = root
    trigger = {
        is_alive = yes
        is_ruler = yes
        dynasty = { has_variable = bp_weles_debt_unpaid_v018 }
    }
    immediate = {
        if = {
            limit = { is_ai = no }
            send_interface_toast = { type = bp_042_divine_arrived_message left_icon = root }
        }
    }
    option = {
        name = bp_weles.2020.a
        trigger = { gold >= 400 }
        show_as_unavailable = { gold < 400 }
        remove_short_term_gold = 400
        add_prestige = -150
        bp_weles_settle_debt_018_effect = yes
        bp_weles_plus_20_effect = yes
        add_character_modifier = { modifier = bp_weles_public_confession_modifier years = 3 }
        ai_chance = { base = 50 }
    }
    option = {
        name = bp_weles.2020.b
        # The ruler survives but accepts a serious, prolonged divine punishment.
        add_stress = 90
        add_prestige = -350
        add_character_modifier = { modifier = bp_weles_debt_shadow_modifier years = 5 }
        bp_weles_minus_15_effect = yes
        ai_chance = { base = 40 }
    }
    option = {
        name = bp_weles.2020.c
        # A fearful ruler suffers for choosing defiance, while the brave
        # may earn renown even when their attempt ends in ruin.
        if = { limit = { has_trait = craven } add_stress = 60 }
        if = { limit = { has_trait = brave } add_prestige = 100 }
        # Deliberately deadly defiance. Chance numbers are prototype values.
        random_list = {
            35 = {
                dynasty = { set_variable = { name = bp_042_weles_wrath_death_ever value = yes } }
                add_character_flag = { flag = bp_weles_divine_death_v030 }
                send_interface_toast = {
                    type = bp_042_trial_failure_message
                    title = bp_042_weles_wrath_dead.t
                    desc = bp_042_weles_wrath_dead.desc
                    left_icon = root
                }
                death = { death_reason = death_execution }
            }
            65 = {
                dynasty = { set_variable = { name = bp_042_weles_wrath_survived_ever value = yes } }
                send_interface_toast = {
                    type = bp_042_trial_failure_message
                    title = bp_042_weles_wrath_survived.t
                    desc = bp_042_weles_wrath_survived.desc
                    left_icon = root
                    add_stress = 100
                    add_prestige = -500
                    add_character_modifier = { modifier = bp_weles_debt_shadow_modifier years = 5 }
                }
                bp_weles_minus_25_effect = yes
            }
        }
