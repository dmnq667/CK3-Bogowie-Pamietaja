# 20-27 July. Perun distinguishes courage from cruelty. Rite, cost, deed.
# 0.4.0: Single holiday scene; divine judgments and blood vengeance are delayed.
bp_perun.9101 = {
    type = character_event
    theme = diplomacy
    override_background = { reference = bp_scene_perun_02 }
    title = bp_041_perun.intro.t
    desc = bp_041_perun.intro.desc
    left_portrait = root
    immediate = {
        # CK3 native message filter requests auto-pause before player choice.
        if = {
            limit = { is_ai = no }
            send_interface_toast = {
                type = bp_042_rite_arrived_message
                left_icon = root
            }
        }
    }
    option = {
        name = bp_041_perun.intro.a
        # Basic festival reward, exactly once per participation.
        add_piety = 45
        add_stress = -10
        add_character_modifier = { modifier = bp_perun_lesser_boon_modifier days = 90 }
        trigger_event = bp_041.2109
        ai_chance = { base = 100 }
    }
}

    desc = bp_perun.9104.desc
    left_portrait = root
    option = {
        name = bp_perun.9104.a
        if = {
            limit = {
                has_character_flag = bp_perun_rite_v019

                has_character_flag = bp_perun_courage_v019

}

# v0.3.0: Perun's anger is a standalone episode, not another mandatory
# ceremonial click. It occurs only after deliberate refusal of his rite.
bp_perun.9110 = {
    type = character_event
    theme = diplomacy
    # ART: Approved 0.3.0 visible-deity scene; never fall back to legacy art.
    override_background = { reference = bp_scene_deity_perun_9110 }
    title = bp_perun.9110.t
    desc = {
        first_valid = {
            triggered_desc = {
                trigger = { has_trait = brave }
                desc = bp_perun.9110.desc_brave
            }
            triggered_desc = {
                trigger = { has_trait = craven }
                desc = bp_perun.9110.desc_craven
            }
            desc = bp_perun.9110.desc_default
        }
    }
    left_portrait = root
    trigger = { is_alive = yes is_ruler = yes }
    immediate = {
        if = {
            limit = { is_ai = no }
            send_interface_toast = { type = bp_042_divine_arrived_message left_icon = root }
        }
    }
    option = {
        name = bp_perun.9110.a
        # A costly public oath protects warriors and reconciles the ruler.
        trigger = { gold >= 150 }
        show_as_unavailable = { gold < 150 }
        remove_short_term_gold = 150
        add_prestige = -150
        add_character_modifier = { modifier = bp_perun_lesser_boon_modifier years = 2 }
        bp_perun_plus_20_effect = yes
        dynasty = { set_variable = { name = bp_perun_reconciled_ever_v030 value = yes } }
        ai_chance = { base = 55 }
    }
    option = {
        name = bp_perun.9110.b
        # Escape with your life but bear the consequences in court and battle.
        add_stress = 60
        add_prestige = -300
        add_character_modifier = { modifier = bp_perun_disfavor_modifier years = 3 }
        bp_perun_minus_20_effect = yes
        ai_chance = { base = 35 }
    }
    option = {
        name = bp_perun.9110.c
        # Warning of mortal danger is explicit in the option text.
        if = { limit = { has_trait = craven } add_stress = 60 }
        if = { limit = { has_trait = brave } add_prestige = 150 }
        random_list = {
            25 = {
                dynasty = { set_variable = { name = bp_042_perun_wrath_death_ever value = yes } }
                send_interface_toast = {
                    type = bp_042_trial_failure_message
                    title = bp_042_perun_wrath_dead.t
                    desc = bp_042_perun_wrath_dead.desc
                    left_icon = root
                }
                death = { death_reason = death_execution }
            }
            75 = {
                dynasty = { set_variable = { name = bp_042_perun_wrath_survived_ever value = yes } }
                send_interface_toast = {
                    type = bp_042_trial_failure_message
                    title = bp_042_perun_wrath_survived.t
                    desc = bp_042_perun_wrath_survived.desc
                    left_icon = root
                    add_stress = 100
                    add_character_modifier = { modifier = bp_perun_disfavor_modifier years = 5 }
                    add_prestige = -400
                }
                bp_perun_minus_20_effect = yes
            }
        }
        ai_chance = { base = 10 }
    }
}
