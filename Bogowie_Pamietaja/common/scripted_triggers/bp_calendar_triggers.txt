# HOTFIX 0.1.3: use expiring dynasty locks rather than current-year script values.
# All four celebration windows are at most seven days long.
# A 335-day lock, set at the moment of initiating the rite, always expires
# before that festival opens next year, but never within the same celebration.
# The lock is scoped to the dynasty so succession does not allow a second rite.
# No numeric year script values are evaluated at startup.

bp_gromnica_calendar_window_trigger = {
    time_of_year = { min = 2.2 max = 2.8 }
}
bp_gromnica_not_celebrated_this_season_trigger = {
    NOT = { has_character_flag = bp_gromnica_used }
    dynasty = { NOT = { has_variable = bp_gromnica_rite_lock_v013 } }
}
bp_gromnica_upcoming_accepted_trigger = {
    dynasty = { has_variable = bp_gromnica_accepted_v013 }
}
