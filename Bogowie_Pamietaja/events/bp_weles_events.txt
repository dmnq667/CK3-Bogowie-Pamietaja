# The opening of Weles Week, 9-15 February.
bp_weles.2001 = {
    type = character_event
    theme = diplomacy
    override_background = {
        reference = bp_scene_weles_04
    }
    title = bp_weles.2001.t
    desc = {
        first_valid = {
            triggered_desc = {
                trigger = { has_character_flag = bp_gromnica_fire }
                desc = bp_weles.2001.desc_fire
            }
            triggered_desc = {
                trigger = { has_character_flag = bp_gromnica_village }
                desc = bp_weles.2001.desc_village
            }
            desc = bp_weles.2001.desc_default
        }
    }
    left_portrait = root
    immediate = {
        bp_initialize_effect = yes
        dynasty = { set_variable = { name = bp_weles_ever_celebrated_v014 value = yes } }
        add_character_flag = { flag = bp_weles_week_used days = 335 }
        dynasty = {
            set_variable = { name = bp_weles_rite_lock_v013 value = yes days = 335 }
            if = {
                limit = { has_variable = bp_weles_accepted_v013 }
                remove_variable = bp_weles_accepted_v013
            }
        }
    }
    option = {
        name = bp_weles.2001.a
        trigger_event = bp_weles.2002
        add_character_flag = { flag = bp_weles_entered_grove_v020 days = 120 }
        ai_chance = { base = 65 }
    }
    option = {
        name = bp_weles.2001.b
        add_character_flag = { flag = bp_weles_fled days = 335 }
        add_character_flag = { flag = bp_weles_fled_now_v020 days = 120 }
        bp_weles_minus_15_effect = yes
        add_prestige = -40
        ai_chance = { base = 35 }
    }
}

# A player explicitly picks which of THEIR living ADULT prisoners to offer.
# The target survives until the choice in this confirmation event.
bp_weles.2004 = {
    type = character_event
    theme = diplomacy
    override_background = {
        reference = bp_scene_weles_03
    }
    title = bp_weles.2004.t
    desc = bp_weles.2004.desc
    left_portrait = root
    right_portrait = scope:bp_offering_person
    trigger = {
        exists = var:bp_weles_proposed_prisoner
        var:bp_weles_proposed_prisoner = {
            is_alive = yes
            is_adult = yes
            is_imprisoned_by = root
        }
    }
    on_trigger_fail = {
        if = {
            limit = { has_variable = bp_weles_proposed_prisoner }
            remove_variable = bp_weles_proposed_prisoner
        }
    }
    immediate = {
        var:bp_weles_proposed_prisoner = { save_scope_as = bp_offering_person }
    }
    option = {
        name = bp_weles.2004.a
        if = {
            limit = { scope:bp_offering_person = { dynasty = root.dynasty } }
            add_character_flag = { flag = bp_weles_dynasty_blood years = 15 }
            dynasty = { set_variable = { name = bp_weles_dynastic_blood_v018 value = yes } }
            add_stress = 60
            add_tyranny = 20
        }
        # Survivors react to a real, named death. This is independent of dynasty status.
        scope:bp_offering_person = {
            every_close_family_member = {
                limit = {
                    is_alive = yes
                    NOT = { this = root }
                }
                add_opinion = { target = root modifier = bp_weles_relative_sacrifice_opinion }
            }
            death = { killer = root death_reason = death_execution }
        }
        remove_character_flag = bp_weles_waiting_for_prisoner
        remove_variable = bp_weles_proposed_prisoner
        add_character_flag = { flag = bp_weles_offering days = 90 }
        add_character_flag = { flag = bp_weles_blood days = 730 }
        add_character_flag = { flag = bp_weles_blood_now_v020 days = 120 }
        bp_weles_plus_20_effect = yes
        add_dread = 15
        trigger_event = bp_weles.2003
        ai_chance = { base = 35 }
    }
    option = {
        name = bp_weles.2004.b
        remove_character_flag = bp_weles_waiting_for_prisoner
        remove_variable = bp_weles_proposed_prisoner
        add_prestige = -80
        add_character_flag = { flag = bp_weles_offering days = 90 }
        add_character_flag = { flag = bp_weles_merciful days = 730 }
        add_character_flag = { flag = bp_weles_mercy_now_v020 days = 120 }
        bp_weles_plus_10_effect = yes
        trigger_event = bp_weles.2003
        ai_chance = { base = 65 }
    }
}
