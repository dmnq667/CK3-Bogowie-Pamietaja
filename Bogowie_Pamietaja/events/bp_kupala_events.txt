bp_kupala.8002 = {
    type = character_event
    theme = diplomacy
    override_background = {
        reference = bp_scene_kupala_02
    }
    title = bp_kupala.8002.t
    desc = {
        first_valid = {
            triggered_desc = {
                trigger = { exists = scope:bp_kupala_companion }
                desc = bp_kupala.8002.desc_person
            }
            desc = bp_kupala.8002.desc_default
        }
    }
    left_portrait = root
    immediate = {
        if = {
            limit = { is_adult = yes }
            random_courtier = {
                limit = {
                    is_adult = yes
                    is_alive = yes
                    can_set_relation_potential_lover_trigger = { CHARACTER = root }
                }
                save_scope_as = bp_kupala_companion
            }
        }
    }
    option = {
        name = bp_kupala.8002.a
        trigger = { is_adult = yes exists = scope:bp_kupala_companion }
        if = {
            limit = { can_set_relation_potential_lover_trigger = { CHARACTER = scope:bp_kupala_companion } }
            set_relation_potential_lover = scope:bp_kupala_companion
        }
        scope:bp_kupala_companion = {
            add_opinion = { target = root modifier = bp_kupala_night_encounter_opinion }
        }
        add_character_flag = { flag = bp_kupala_desire_trial days = 90 }
        add_character_flag = { flag = bp_kupala_secret_night years = 2 }
        trigger_event = bp_kupala.8003
        ai_chance = { base = 35 }
    }
    option = {
        name = bp_kupala.8002.b
        add_character_flag = { flag = bp_kupala_desire_trial days = 90 }
        add_character_flag = { flag = bp_kupala_faithful_night years = 2 }
        bp_mokosh_plus_10_effect = yes
        trigger_event = bp_kupala.8003
        ai_chance = { base = 50 }
    }
    option = {
        name = bp_kupala.8002.c
        add_prestige = -20
        trigger_event = bp_kupala.8003
        ai_chance = { base = 15 }
    }
}

            }
            desc = bp_kupala.8003.desc
        }
    }
    left_portrait = root
    option = {
        name = bp_kupala.8003.a
        trigger = { gold >= 40 }

        remove_short_term_gold = 40
        add_character_flag = { flag = bp_kupala_truth_trial days = 90 }

            limit = {
                has_character_flag = bp_kupala_fire_trial
                has_character_flag = bp_kupala_desire_trial
                is_adult = yes
                NOT = { has_character_flag = bp_kupala_great_received }

        ai_chance = { base = 50 }
