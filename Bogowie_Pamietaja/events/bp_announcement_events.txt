    type = character_event
    # Legacy opt-in notice becomes a SILENT router. The educational
    # opening and the dramatic chapter are the only two visible modals.
    hidden = yes
    trigger = {
        has_character_flag = bp_verbose_announcements_optin_v021
        is_alive = yes
        is_ruler = yes
        faith = { religion_tag = slavic_religion }
        bp_gromnica_calendar_window_trigger = yes

            NOT = { has_variable = bp_gromnica_announced_v014 }
            NOT = { has_variable = bp_gromnica_rite_lock_v013 }

    immediate = {
        dynasty = {
            set_variable = { name = bp_gromnica_announced_v014 value = yes days = 335 }
        }
        trigger_event = bp_gromnica.1001
    }
}
