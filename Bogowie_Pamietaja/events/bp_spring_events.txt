namespace = bp_spring

# 9 March. Mokosh enters as a distinct, powerful god with her own values.
bp_spring.3001 = {
    type = character_event
    theme = diplomacy
    override_background = {
        reference = bp_scene_spring_01
    }
    title = bp_spring.3001.t
    desc = {
        first_valid = {
            triggered_desc = {
                trigger = { OR = { has_character_flag = bp_weles_blood dynasty = { has_variable = bp_weles_dynastic_blood_v018 } } }
                desc = bp_spring.3001.desc_blood
            }
            triggered_desc = {
                trigger = { OR = { has_character_flag = bp_weles_debt_years dynasty = { has_variable = bp_weles_debt_unpaid_v018 } } }
                desc = bp_spring.3001.desc_debt
            }
            desc = bp_spring.3001.desc
        }
    }
    left_portrait = root
    immediate = {
        bp_initialize_effect = yes
        add_piety = 50
        add_stress = -10
        dynasty = { set_variable = { name = bp_spring_ever_celebrated_v014 value = yes } }
        add_character_flag = { flag = bp_spring_used days = 335 }
        dynasty = {
            set_variable = { name = bp_spring_rite_lock_v013 value = yes days = 335 }
            if = {
                limit = { has_variable = bp_spring_accepted_v013 }
                remove_variable = bp_spring_accepted_v013
            }
        }
    }
    option = {
        name = bp_spring.3001.a
        trigger = { gold >= 40 }
        show_as_unavailable = { gold < 40 }
        remove_short_term_gold = 40
        add_character_flag = { flag = bp_mokosh_rite days = 90 }
        bp_mokosh_plus_20_effect = yes
        capital_county = {
            add_county_modifier = { modifier = bp_mokosh_sowing_modifier years = 2 }
        }
        trigger_event = bp_spring.3002
        ai_chance = { base = 50 }
    }
    option = {
        name = bp_spring.3001.b
        add_prestige = -50
        add_character_flag = { flag = bp_mokosh_rite days = 90 }
        bp_mokosh_plus_20_effect = yes
        trigger_event = bp_spring.3002
        ai_chance = { base = 40 }
    }
    option = {
        name = bp_spring.3001.c
        add_gold = 25
        bp_mokosh_minus_15_effect = yes
        dynasty = { set_variable = { name = bp_mokosh_sowing_neglected_ever_v030 value = yes } }
        # The goddess does not interrupt every rite, but can return later.
        trigger_event = { id = bp_spring.3010 days = { 45 75 } }
        ai_chance = { base = 10 }
    }
}

# v0.3.0 Mokosh's intervention, distinct from ordinary festival scenes.
# A direct consequence of withholding grain during Spring New Year.
bp_spring.3010 = {
    type = character_event
    theme = diplomacy
    # ART: Approved 0.3.0 visible-deity scene; never fall back to legacy art.
    override_background = { reference = bp_scene_deity_mokosh_3010 }
    title = bp_spring.3010.t
    desc = {
        first_valid = {
            triggered_desc = { trigger = { has_trait = compassionate } desc = bp_spring.3010.desc_compassionate }
            triggered_desc = { trigger = { has_trait = callous } desc = bp_spring.3010.desc_callous }
            desc = bp_spring.3010.desc_default
        }
    }
    left_portrait = root
    trigger = { is_alive = yes is_ruler = yes }
    option = {
        name = bp_spring.3010.a
        trigger = { gold >= 100 }
        show_as_unavailable = { gold < 100 }
        remove_short_term_gold = 100
        capital_county = { add_county_modifier = { modifier = bp_mokosh_sowing_modifier years = 2 } }
        bp_mokosh_plus_20_effect = yes
        dynasty = { set_variable = { name = bp_mokosh_restitution_ever_v030 value = yes } }
        ai_chance = { base = 55 }
    }
    option = {
        name = bp_spring.3010.b
        # Admit the wrong in public and give up the ruler's renown.
        add_prestige = -250
        add_stress = 40
        bp_mokosh_plus_10_effect = yes
        ai_chance = { base = 35 }
    }
    option = {
        name = bp_spring.3010.c
        # Defying the goddess makes the famine worse, not a random death.
        capital_county = { add_county_modifier = { modifier = bp_harvest_blight_modifier years = 3 } }
        add_stress = 60
        bp_mokosh_minus_15_effect = yes
        ai_chance = { base = 10 }
    }
}
