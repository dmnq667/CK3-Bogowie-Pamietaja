bp_harvest.9201 = {
    type = character_event
    theme = diplomacy
    override_background = { reference = bp_scene_harvest_01 }
    title = bp_harvest.9201.t
    desc = {
        first_valid = {
            triggered_desc = {
                trigger = { capital_county = { has_county_modifier = bp_gaik_wounded_land_modifier } }
                desc = bp_harvest.9201.desc_hard
            }
            desc = bp_harvest.9201.desc
        }
    }
    left_portrait = root
    immediate = {
        bp_initialize_effect = yes
        dynasty = {
            set_variable = { name = bp_harvest_ever_celebrated_v019 value = yes }
            set_variable = { name = bp_harvest_rite_lock_v019 value = yes days = 335 }
        }
    }
    option = {
        name = bp_harvest.9201.a
        trigger = { gold >= 60 }
        show_as_unavailable = { gold < 60 }
        remove_short_term_gold = 60
        add_character_flag = { flag = bp_harvest_rite_v019 days = 120 }
        bp_mokosh_plus_20_effect = yes
        trigger_event = bp_harvest.9202
        ai_chance = { base = 50 }
    }
    option = {
        name = bp_harvest.9201.b
        add_prestige = -100
        add_character_flag = { flag = bp_harvest_rite_v019 days = 120 }
        bp_mokosh_plus_10_effect = yes
        trigger_event = bp_harvest.9202
        ai_chance = { base = 35 }
    }
    option = {
        name = bp_harvest.9201.c
        bp_mokosh_minus_15_effect = yes
        trigger_event = bp_harvest.9206
        ai_chance = { base = 15 }
    }
}
