namespace = bp_szczodre

# One sacred season spans December 21 through January 6. A 335-day dynasty
# lock placed when the rite starts (or is declined) prevents a second rite on
# January 1 and expires before the next December 21.
bp_szczodre.9401 = {
    type = character_event
    theme = diplomacy
    override_background = { reference = bp_scene_szczodre_01 }
    title = bp_szczodre.9401.t
    desc = bp_szczodre.9401.desc
    left_portrait = root
    immediate = {
        bp_initialize_effect = yes
        dynasty = {
            set_variable = { name = bp_szczodre_ever_celebrated_v019 value = yes }
            set_variable = { name = bp_szczodre_rite_lock_v019 value = yes days = 335 }
        }
    }
    option = {
        name = bp_szczodre.9401.a
        trigger = { gold >= 60 }
        show_as_unavailable = { gold < 60 }
        remove_short_term_gold = 60
        add_character_flag = { flag = bp_szczodre_fire_v019 days = 90 }
        bp_mokosh_plus_20_effect = yes
        trigger_event = bp_szczodre.9402
        ai_chance = { base = 50 }
    }
    option = {
        name = bp_szczodre.9401.b
        add_prestige = -100
        add_character_flag = { flag = bp_szczodre_fire_v019 days = 90 }
        bp_mokosh_plus_10_effect = yes
        trigger_event = bp_szczodre.9402
        ai_chance = { base = 35 }
    }
    option = {
        name = bp_szczodre.9401.c
        bp_mokosh_minus_15_effect = yes
        trigger_event = bp_szczodre.9407
        ai_chance = { base = 15 }
    }
}
