# The blessing requires all three independent character flags; no free bonus.
bp_perun.9104 = {
    type = character_event
    theme = diplomacy
    override_background = { reference = bp_scene_perun_03 }
    title = bp_perun.9104.t
    desc = bp_perun.9104.desc
    left_portrait = root
    option = {
        name = bp_perun.9104.a
        if = {
            limit = {
                has_character_flag = bp_perun_rite_v019
                has_character_flag = bp_perun_price_v019
                has_character_flag = bp_perun_courage_v019
                has_character_flag = bp_perun_mercy_v019
                NOT = { has_character_flag = bp_perun_great_received_v019 }
            }
            trigger_event = bp_perun.9105
        }
        else = { trigger_event = bp_perun.9106 }
        ai_chance = { base = 100 }
    }
}

bp_perun.9105 = {
    type = character_event
    theme = diplomacy
    override_background = { reference = bp_scene_perun_03 }
    title = bp_perun.9105.t
    desc = {
        first_valid = {
            triggered_desc = { trigger = { is_at_war = yes } desc = bp_perun.9105.desc_war }
            desc = bp_perun.9105.desc_peace
        }
    }
    left_portrait = root
    option = {
        name = bp_perun.9105.a
        if = {
            limit = { is_at_war = yes }
            add_character_modifier = { modifier = bp_perun_great_war_modifier years = 1 }
            add_character_flag = { flag = bp_perun_boon_active_v019 years = 1 }
        }
        else = { add_character_flag = { flag = bp_perun_boon_reserved_v019 years = 2 } }
        add_character_flag = { flag = bp_perun_great_received_v019 years = 3 }
        dynasty = { set_variable = { name = bp_perun_ever_blessed_v019 value = yes } }
        bp_perun_plus_40_effect = yes
        trigger_event = { id = bp_perun.9108 days = { 100 180 } }
        ai_chance = { base = 100 }
    }
}
