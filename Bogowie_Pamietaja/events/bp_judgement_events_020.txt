namespace = bp_judgement

# COURT I: Popiół i Przysięga, once every winter after Weles Week.
# One existing watcher schedules the event, do not add another on_action.
# The dynasty lock blocks repeats via reload, succession and debug.
bp_judgement.1001 = {
    type = character_event
    theme = diplomacy
    override_background = { reference = bp_scene_gromnica_02 }
    title = bp_judgement.1001.t
    desc = {
        first_valid = {
            triggered_desc = {
                trigger = { has_character_flag = bp_gromnica_both_done_v020 }
                desc = bp_judgement.1001.desc_both
            }
            triggered_desc = {
                trigger = { has_character_flag = bp_gromnica_ignored_v020 }
                desc = bp_judgement.1001.desc_ignored
            }
            triggered_desc = {
                trigger = { has_character_flag = bp_gromnica_fire }
                desc = bp_judgement.1001.desc_fire
            }
            triggered_desc = {
                trigger = { has_character_flag = bp_gromnica_village }
                desc = bp_judgement.1001.desc_village
            }
            desc = bp_judgement.1001.desc_none
        }
    }
    left_portrait = root
    trigger = {
        is_alive = yes
        is_ruler = yes
        is_ai = no
        faith = { religion_tag = slavic_religion }
        dynasty = { NOT = { has_variable = bp_first_court_lock_v020 } }
    }
    immediate = {
        add_character_flag = { flag = bp_first_court_started_v020 days = 345 }
        dynasty = { set_variable = { name = bp_first_court_lock_v020 value = yes days = 345 } }
    }
    option = {
        name = bp_judgement.1001.a
        trigger_event = bp_judgement.1002
        ai_chance = { base = 100 }
    }
}
