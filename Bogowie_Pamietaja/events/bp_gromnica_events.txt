# 2 February; initiated through the decision during the correct date window.
bp_gromnica.1001 = {

    theme = diplomacy
    override_background = {
        reference = bp_scene_gromnica_02
    }
    title = bp_gromnica.1001.t
    desc = bp_gromnica.1001.desc
    left_portrait = root
    immediate = {
        bp_initialize_effect = yes
        dynasty = { set_variable = { name = bp_gromnica_ever_celebrated_v014 value = yes } }

        dynasty = {

        bp_mokosh_plus_10_effect = yes

    }
    option = {
        name = bp_gromnica.1001.b
        add_prestige = -50
        add_character_flag = { flag = bp_gromnica_trial days = 90 }
        add_character_flag = { flag = bp_gromnica_village days = 90 }
        bp_weles_plus_10_effect = yes
        bp_mokosh_plus_20_effect = yes
        trigger_event = { id = bp_gromnica.1002 days = 1 }
        ai_chance = { base = 35 }
    }
    option = {
        name = bp_gromnica.1001.c
        add_gold = 20
        add_character_flag = { flag = bp_gromnica_ignored_v020 days = 335 }
        bp_weles_minus_15_effect = yes
        trigger_event = { id = bp_gromnica.1002 days = 1 }
        ai_chance = { base = 10 }
    }
}

# This is a genuine second scene, not a tooltip on the first event.
bp_gromnica.1002 = {
    type = character_event
    theme = diplomacy
    override_background = {
        trigger = { has_character_flag = bp_gromnica_fire }
        reference = bp_scene_gromnica_01
    }
    override_background = {
        reference = bp_scene_gromnica_02
    }
    title = bp_gromnica.1002.t
    desc = {
        first_valid = {
            triggered_desc = {
                trigger = { has_character_flag = bp_gromnica_fire }
                desc = bp_gromnica.1002.desc_fire
            }
            triggered_desc = {
                trigger = { has_character_flag = bp_gromnica_village }
                desc = bp_gromnica.1002.desc_village
            }
            desc = bp_gromnica.1002.desc_ignored
        }
    }
    left_portrait = root
    # v0.2.0: the player may complete BOTH Gromnica duties in the second scene.
    # First fire, then village relief. This must be a real second sacrifice.
    option = {
        name = bp_gromnica.1002.b
        trigger = {
            has_character_flag = bp_gromnica_fire
            NOT = { has_character_flag = bp_gromnica_village }
        }
        add_prestige = -50
        add_character_flag = { flag = bp_gromnica_village days = 90 }
        add_character_flag = { flag = bp_gromnica_both_done_v020 days = 335 }
        add_character_flag = { flag = bp_gromnica_second_village_v020 days = 90 }
        bp_mokosh_plus_10_effect = yes
        trigger_event = bp_gromnica.1003
        ai_chance = { base = 35 }
    }
    # First village relief, then keep the gate fires and guards alive.
    option = {
        name = bp_gromnica.1002.c
        trigger = {
            has_character_flag = bp_gromnica_village
            NOT = { has_character_flag = bp_gromnica_fire }
            gold >= 35
        }
        show_as_unavailable = {
            has_character_flag = bp_gromnica_village
            NOT = { has_character_flag = bp_gromnica_fire }
            gold < 35
        }
        remove_short_term_gold = 35
        add_character_flag = { flag = bp_gromnica_fire days = 90 }
        add_character_flag = { flag = bp_gromnica_both_done_v020 days = 335 }
        add_character_flag = { flag = bp_gromnica_second_fire_v020 days = 90 }
        add_character_modifier = { modifier = bp_gromnica_guard_modifier days = 45 }
        bp_weles_plus_10_effect = yes
        trigger_event = bp_gromnica.1003
        ai_chance = { base = 35 }
    }
    # Preserve the original option for players who do not take a second action.
    option = {
        name = bp_gromnica.1002.a
        add_character_flag = { flag = bp_heard_ice_voice days = 90 }
        ai_chance = { base = 100 }
    }
}
