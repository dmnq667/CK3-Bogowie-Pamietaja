namespace = bp_calendar

# All four festivals share a year-round decision surface.
# These events never award boons and never mark a ritual as completed.

# gromnica: player registers/cancels an upcoming holiday, with no ritual effect.
bp_calendar.8101 = {
    type = character_event
    theme = diplomacy
    override_background = {
        reference = bp_scene_gromnica_02
    }
    title = bp_calendar.8101.t
    desc = {
        first_valid = {
            triggered_desc = {
                trigger = { bp_gromnica_upcoming_accepted_trigger = yes }
                desc = bp_calendar.8101.desc_registered
            }
            desc = bp_calendar.8101.desc
        }
    }
    left_portrait = root
    option = {
        name = bp_calendar.accept_next
        trigger = { NOT = { bp_gromnica_upcoming_accepted_trigger = yes } }
        dynasty = {
            set_variable = {
                name = bp_gromnica_accepted_v013
                value = yes
                days = 365
            }
        }
        ai_chance = { base = 0 }
    }
    option = {
        name = bp_calendar.cancel_next
        trigger = { bp_gromnica_upcoming_accepted_trigger = yes }
        dynasty = { remove_variable = bp_gromnica_accepted_v013 }
        ai_chance = { base = 0 }
    }
    option = {
        name = bp_calendar.close
        ai_chance = { base = 100 }
    }
}
