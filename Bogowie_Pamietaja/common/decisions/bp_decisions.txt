# v0.1.2: persistent decisions with conditional in-window ceremony / future reservation.
# The book is always reusable with no cost.
bp_book_decision = {
    picture = { reference = "gfx/interface/illustrations/decisions/decision_personal_religious.dds" }
    desc = bp_book_decision_desc
    selection_tooltip = bp_book_decision_tooltip
    is_shown = {
        is_ai = no
        is_ruler = yes
        OR = {
            faith = { religion_tag = slavic_religion }
            has_variable = bp_weles_favor
            has_variable = bp_mokosh_favor
            has_character_flag = bp_weles_debt_years
            dynasty = { has_variable = bp_weles_debt_unpaid_v018 }
            dynasty = { has_variable = bp_weles_debt_paid_v018 }
        }
    }
    is_valid = { always = yes }
    should_create_alert = { always = no }
    effect = {
        bp_initialize_effect = yes
        bp_weles_migrate_legacy_debt_018_effect = yes
        bp_start_calendar_watch_016_effect = yes
        trigger_event = bp_book.0001
    }
    ai_check_interval = 0
}

# Let the player resume the Weles ritual if no suitable prisoner exists.
bp_weles_change_offering_decision = {
    picture = { reference = "gfx/interface/illustrations/decisions/decision_personal_religious.dds" }
    desc = bp_weles_change_offering_decision_desc
    selection_tooltip = bp_weles_change_offering_decision_tooltip
    is_shown = {
        is_ai = no
        has_character_flag = bp_weles_waiting_for_prisoner
    }
    is_valid = { always = yes }
    should_create_alert = { always = no }
    effect = {
        remove_character_flag = bp_weles_waiting_for_prisoner
        if = {
            limit = { has_variable = bp_weles_proposed_prisoner }
            remove_variable = bp_weles_proposed_prisoner
        }
        trigger_event = bp_weles.2002
    }
    ai_check_interval = 0
}

# An unpaid dynastic debt can be settled without waiting a full year.
# This is not a holiday and does not consume an annual rite.
bp_weles_reckon_debt_decision = {
    picture = { reference = "gfx/interface/illustrations/decisions/decision_personal_religious.dds" }
    desc = bp_weles_reckon_debt_decision_desc
    selection_tooltip = bp_weles_reckon_debt_decision_tooltip
    is_shown = {
        is_ai = no
        is_ruler = yes
        dynasty = { has_variable = bp_weles_debt_unpaid_v018 }
        NOT = { has_character_flag = bp_weles_reckoning_recent_v018 }
    }
    is_valid = { always = yes }
    should_create_alert = { always = no }
    effect = {
        add_character_flag = { flag = bp_weles_reckoning_recent_v018 days = 90 }
        trigger_event = bp_weles.2013
    }
    ai_check_interval = 0
}

# Perun's earned boon can be invoked in a later war; not a new holiday.
bp_perun_call_storm_decision = {
    picture = { reference = "gfx/interface/illustrations/decisions/decision_personal_religious.dds" }
    desc = bp_perun_call_storm_decision_desc
    selection_tooltip = bp_perun_call_storm_decision_tooltip
    confirm_text = bp_perun_call_storm_decision_confirm
    decision_group_type = major
    is_shown = {
        is_ai = no
        is_ruler = yes
        has_character_flag = bp_perun_boon_reserved_v019
    }
    is_valid = { is_at_war = yes }
    should_create_alert = { always = yes }
    effect = {
        remove_character_flag = bp_perun_boon_reserved_v019
        add_character_modifier = { modifier = bp_perun_great_war_modifier years = 1 }
        add_character_flag = { flag = bp_perun_boon_active_v019 years = 1 }
    }
    ai_check_interval = 0
}
